# HMM: keep this in sync with the on_war_relation_added block in on actions responsible for removing the no grind trait
can_grind_generals = {
	if = {
		limit = { axis_european = yes }
		has_global_flag = axis_grind_allowed
		OR = {
			has_war_with = ENG
			has_war_with = USA
			has_war_with = SOV
			tag = SPR
		}
	}
	else_if = {
		limit = { axis_geacps = yes }
		has_global_flag = geacps_grind_allowed
		OR = {
			has_war_with = ENG
			has_war_with = USA
			has_war_with = SOV # Just in case
		}
	}
	else_if = {
		limit = { allies = yes }
		has_global_flag = allies_grind_allowed
		OR = {
			has_war_with = GER
			has_war_with = JAP
		}
	}
}

ai_annex = {
	# We need this giant if-else structure because if we just use ANDs, the game crashes on startup
	if = {
		limit = {
			has_game_rule = {
				rule = baltics_status
				option = abstracted
			}
		}
		if = {
			limit = {
				has_game_rule = {
					rule = austria_status
					option = abstracted
				}
			}
			if = {
				limit = {
					has_game_rule = {
						rule = czechs_status
						option = abstracted
					}
				}
				OR = {
					is_baltic_country = yes
					tag = AUS
					tag = CZE
					tag = COG
				}
			}
			else = {
				OR = {
					is_baltic_country = yes
					tag = AUS
					tag = COG
				}
			}
		}
		else_if = {
			limit = {
				has_game_rule = {
					rule = czechs_status
					option = abstracted
				}
			}
			OR = {
				is_baltic_country = yes
				tag = CZE
				tag = COG
			}
		}
		else = {
			OR = {
				is_baltic_country = yes
				tag = COG
			}
		}
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = austria_status
				option = abstracted
			}
		}
		if = {
			limit = {
				has_game_rule = {
					rule = czechs_status
					option = abstracted
				}
			}
			OR = {
				tag = AUS
				tag = CZE
				tag = COG
			}
		}
		else = {
			OR = {
				tag = AUS
				tag = COG
			}
		}
	}
	else_if = {
		limit = {
			has_game_rule = {
				rule = czechs_status
				option = abstracted
			}
		}
		OR = {
			tag = CZE
			tag = COG
		}
	}
	else = {
		tag = COG
	}
}
