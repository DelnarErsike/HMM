on_actions = {
	on_startup = {
		effect = {
			# Give every country free infiltrations on all other countries in their same team
			# We want people to be able to see what other people on their team are doing without having to ask them, even if they're not in the same faction
			# This could also be done by having an on monthly effect that keeps adding intel, but that's too taxing for the effect we want to have, and this is good enough
			every_country = {
				if = {
					limit = { allies = yes }
					every_other_country = {
						limit = { allies = yes }
						add_operation_token = {
							tag = PREV
							token = token_civilian
						}
						add_operation_token = {
							tag = PREV
							token = token_army
						}
						add_operation_token = {
							tag = PREV
							token = token_navy
						}
						add_operation_token = {
							tag = PREV
							token = token_airforce
						}
					}
				}
				else_if = {
					limit = { axis = yes }
					every_other_country = {
						limit = { axis = yes }
						add_operation_token = {
							tag = PREV
							token = token_civilian
						}
						add_operation_token = {
							tag = PREV
							token = token_army
						}
						add_operation_token = {
							tag = PREV
							token = token_navy
						}
						add_operation_token = {
							tag = PREV
							token = token_airforce
						}
					}
				}
			}
		}
	}
}
