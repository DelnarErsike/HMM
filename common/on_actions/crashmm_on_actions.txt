    on_daily = {
        effect = {
            every_country = {
                every_army_leader = {
					remove_unit_leader_trait = no_grind
				}
            }
        }
    }

    on_weekly = {
        effect = {
            every_country = {

					add_unit_leader_trait = no_grind
				}
            }
        }
    }
}
